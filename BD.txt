CREATE KEYSPACE energia
WITH replication = {
  'class':'SimpleStrategy',
  'replication_factor':1
};


USE energia;


CREATE TABLE consumos (

    id_dispositivo text,
    fecha timestamp,

    zona text,

    potencia double,
    voltaje double,
    corriente double,

    temperatura double,

    movimiento boolean,

    estado text,

    PRIMARY KEY (id_dispositivo, fecha)

) WITH CLUSTERING ORDER BY (fecha DESC);
